set -eu
root=/home/dev/rs-l2tpd
candidate=$(cat /tmp/phase6-resolver-validation-path)
record=$root/audit/phase6/signal-and-ownership-guards-20261102
cp "$root/src/bin/rs-l2tpd/signals.rs" "$candidate/src/bin/rs-l2tpd/signals.rs"
cp "$root/src/reconciler/tests.rs" "$candidate/src/reconciler/tests.rs"

}
trap restore 0
cd "$candidate"
