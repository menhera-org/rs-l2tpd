set -eu
record=$PWD/audit/phase6/event-and-lifetime-ordering-20261102/kernels
run() {
 label=$1; image=$2; modules=$3; config=$4; mode=$5
 result=0
 sh tests/kernel/run-guest.sh "$image" "$modules" "$config" "$mode" "$record/$label" > "/tmp/phase6-ordering-$label.log" 2>&1 || result=$?
 printf '%s\n' "$result" > "$record/$label.exit"
}
noble=$PWD/target/phase6-kernels/ubuntu-noble-6.8.0-147.147/root
resolute=$PWD/target/phase6-kernels/ubuntu-resolute-7.0.0-42.42/root
run noble-clock "$noble/boot/vmlinuz-6.8.0-147-generic" "$noble/lib/modules/6.8.0-147-generic" "$noble/boot/config-6.8.0-147-generic" clock &
a=$!
run resolute-clock "$resolute/boot/vmlinuz-7.0.0-42-generic" "$resolute/usr/lib/modules/7.0.0-42-generic" "$resolute/boot/config-7.0.0-42-generic" clock &
b=$!
pids="$a $b"
for version in 6.8.12 7.0.14 7.4-rc1; do
 root=$PWD/target/phase6-kernels/linux-$version-ipv6
 run "vanilla-$version-clock" "$root/arch/x86/boot/bzImage" - "$root/.config" clock &
 pids="$pids $!"
done
root=$PWD/target/phase6-kernels/linux-7.4-rc1-ipv6
run ipv6-disable "$root/arch/x86/boot/bzImage" - "$root/.config" ipv6-disable &
pids="$pids $!"
root=$PWD/target/phase6-kernels/linux-7.4-rc1-no-ipv6
run no-ipv6 "$root/arch/x86/boot/bzImage" - "$root/.config" no-ipv6 &
pids="$pids $!"
run no-ethernet "$resolute/boot/vmlinuz-7.0.0-42-generic" "$resolute/usr/lib/modules/7.0.0-42-generic" "$resolute/boot/config-7.0.0-42-generic" no-ethernet &
pids="$pids $!"
for pid in $pids; do wait "$pid"; done
for result in "$record"/*-clock.exit "$record"/ipv6-disable.exit "$record"/no-ipv6.exit "$record"/no-ethernet.exit; do
 printf '%s: ' "${result##*/}"
 cat "$result"
done
