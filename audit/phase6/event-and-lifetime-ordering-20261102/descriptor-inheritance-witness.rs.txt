use std::io::{Read, Write};
use std::os::fd::AsRawFd;
use std::os::unix::{net::UnixStream, process::CommandExt};
use std::process::Command;

unsafe extern "C" {
    fn read(fd: i32, buffer: *mut std::ffi::c_void, count: usize) -> isize;
    fn write(fd: i32, buffer: *const std::ffi::c_void, count: usize) -> isize;
}

fn main() {
    let (mut observer, owner) = UnixStream::pair().unwrap();
    observer.set_nonblocking(true).unwrap();
    let (mut controller, child_control) = UnixStream::pair().unwrap();
    controller.set_read_timeout(Some(std::time::Duration::from_secs(5))).unwrap();
    let launcher = std::thread::spawn(move || {
        let descriptor = child_control.as_raw_fd();
        let mut command = Command::new("/bin/true");
        unsafe {
            command.pre_exec(move || {
                let mut byte = 1_u8;
                if write(descriptor, (&byte as *const u8).cast(), 1) != 1
                    || read(descriptor, (&mut byte as *mut u8).cast(), 1) != 1
                {
                    return Err(std::io::Error::last_os_error());
                }
                Ok(())
            });
        }
        let mut child = command.spawn().unwrap();
        assert!(child.wait().unwrap().success());
    });
    let mut byte = [0];
    controller.read_exact(&mut byte).unwrap();
    assert_eq!(byte, [1]);
    drop(owner);
    assert_eq!(
        observer.read(&mut byte).unwrap_err().kind(),
        std::io::ErrorKind::WouldBlock,
        "the child must retain the closed parent descriptor before exec"
    );
    controller.write_all(&[1]).unwrap();
    launcher.join().unwrap();
    assert_eq!(observer.read(&mut byte).unwrap(), 0);
    println!("CHILD_INHERITANCE_UNTIL_EXEC_CHECKED=1");
}
