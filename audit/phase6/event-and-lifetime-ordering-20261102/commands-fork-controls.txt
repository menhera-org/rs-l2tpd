set -eu
root=/home/dev/rs-l2tpd
record=$root/audit/phase6/event-and-lifetime-ordering-20261102
candidate=$(cat /tmp/phase6-resolver-validation-path)
export CARGO_BUILD_JOBS=2
cd "$candidate"

cp "$root/src/bin/rs-l2tpd/daemon.rs" src/bin/rs-l2tpd/daemon.rs
/tmp/phase6-replace src/bin/rs-l2tpd/daemon.rs < "$record/control-first-fork.txt"
git --no-pager diff -- src/bin/rs-l2tpd/daemon.rs > "$record/control-first-fork.patch"
cargo build --locked --all-features --target x86_64-unknown-linux-gnu > "$record/control-first-fork-build.log" 2>&1
result=0
sh "$root/tests/docker/thread-limits.sh" nd-test:u2404 "$candidate/target/x86_64-unknown-linux-gnu/debug/rs-l2tpd" "$record/control-first-fork" "$candidate/target/fork-control-profiles" > "$record/control-first-fork.log" 2>&1 || result=$?
cp "$root/src/bin/rs-l2tpd/daemon.rs" src/bin/rs-l2tpd/daemon.rs
[ "$result" != 0 ]
grep -q 'panicked' "$record/control-first-fork.log"
printf 'ORDERING_FORK_NEGATIVE_CONTROL=first-fork\n'
