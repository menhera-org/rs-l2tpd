set -eu
record=$PWD/audit/phase6/event-and-lifetime-ordering-20261102
mv "$record/kernels" "$record/rejected-single-thread-one-cpu"
mkdir "$record/kernels"
for label in noble resolute vanilla-6.8.12 vanilla-7.0.14 vanilla-7.4-rc1 ipv6-disable no-ipv6 no-ethernet; do
 [ "$(cat "$record/rejected-parallel-kernels/$label.exit")" = 0 ]
 cp -Rp "$record/rejected-parallel-kernels/$label" "$record/kernels/$label"
 cp "$record/rejected-parallel-kernels/$label.exit" "$record/kernels/$label.exit"
done
run() {
 label=$1; image=$2; modules=$3; config=$4
 result=0
 sh tests/kernel/run-guest.sh "$image" "$modules" "$config" clock "$record/kernels/$label" > "$record/$label-wrapper.log" 2>&1 || result=$?
 printf '%s\n' "$result" > "$record/kernels/$label.exit"
}
noble=$PWD/target/phase6-kernels/ubuntu-noble-6.8.0-147.147/root
resolute=$PWD/target/phase6-kernels/ubuntu-resolute-7.0.0-42.42/root
run noble-clock "$noble/boot/vmlinuz-6.8.0-147-generic" "$noble/lib/modules/6.8.0-147-generic" "$noble/boot/config-6.8.0-147-generic"
run resolute-clock "$resolute/boot/vmlinuz-7.0.0-42-generic" "$resolute/usr/lib/modules/7.0.0-42-generic" "$resolute/boot/config-7.0.0-42-generic"
for version in 6.8.12 7.0.14 7.4-rc1; do
 root=$PWD/target/phase6-kernels/linux-$version-ipv6
 run "vanilla-$version-clock" "$root/arch/x86/boot/bzImage" - "$root/.config"
done
for result in "$record"/kernels/*-clock.exit; do
 printf '%s=' "${result##*/}"
