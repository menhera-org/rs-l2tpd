set -eu
root=/home/dev/rs-l2tpd
candidate=$(cat /tmp/phase6-resolver-validation-path)
record=$root/audit/phase6/probe-and-event-boundaries-20261102
restore() {
 cp /tmp/phase6-probe-control-restart.rs "$candidate/src/bin/rs-l2tpd/restart.rs"
 cp /tmp/phase6-probe-control-signals.rs "$candidate/src/bin/rs-l2tpd/signals.rs"
 cp /tmp/phase6-probe-control-sessions.rs "$candidate/src/reconciler/sessions.rs"
}
trap restore 0
cd "$candidate"
# Remove the final drain limit while retaining the earlier read limit.
awk '/\/\/ Drain the small remaining pipe payload/ {drain=1} drain && /if output.len\(\) > VERSION.len\(\)/ {sub(/if output.len\(\) > VERSION.len\(\)/, "if false"); drain=0} {print}' /tmp/phase6-probe-control-restart.rs > src/bin/rs-l2tpd/restart.rs
diff -u /tmp/phase6-probe-control-restart.rs src/bin/rs-l2tpd/restart.rs > "$record/control-drain-limit.patch" || [ "$?" = 1 ]
if cargo test --locked --all-features --bin rs-l2tpd probe_drains_output -- --nocapture > "$record/control-drain-limit.log" 2>&1; then echo 'drain control unexpectedly passed'; exit 1; fi
grep -q 'reads=3' "$record/control-drain-limit.log"
restore
sed 's/collector_shared.observe(Signal::SIGTERM);/\/\/ controlled omission of the shutdown request/' /tmp/phase6-probe-control-signals.rs > src/bin/rs-l2tpd/signals.rs
diff -u /tmp/phase6-probe-control-signals.rs src/bin/rs-l2tpd/signals.rs > "$record/control-collector-shutdown.patch" || [ "$?" = 1 ]
if cargo test --locked --all-features --bin rs-l2tpd blocked_signals_and_wait_failures -- --nocapture > "$record/control-collector-shutdown.log" 2>&1; then echo 'collector control unexpectedly passed'; exit 1; fi
grep -q 'collector did not request shutdown' "$record/control-collector-shutdown.log"
restore
sed 's/record.name_state == Enforce::Applied \&\& matches!(record.state, Identity::Proved(_))/record.name_state != Enforce::Overridden \&\& record.state.registration().is_some()/' /tmp/phase6-probe-control-sessions.rs > src/reconciler/sessions.rs
diff -u /tmp/phase6-probe-control-sessions.rs src/reconciler/sessions.rs > "$record/control-staged-rename.patch" || [ "$?" = 1 ]
if cargo test --locked --all-features --lib a_name_observed_during_staging -- --nocapture > "$record/control-staged-rename.log" 2>&1; then echo 'staged rename control unexpectedly passed'; exit 1; fi
grep -q 'Overridden' "$record/control-staged-rename.log"
restore
printf 'THREE_PROBE_AND_EVENT_FAILURE_CONTROLS_RESULT=0\n'
