
#[test]
#[ignore = "disposable privileged container or guest with a private writable cgroup2 hierarchy"]
fn both_daemon_forks_report_task_limit_failures_without_starting_service_threads() {
    assert_eq!(
        fs::read_to_string("/proc/self/cgroup").unwrap().trim(),
        "0::/",
        "requires a private cgroup namespace"
    );
    let cgroups = Path::new("/sys/fs/cgroup");
    assert!(
        fs::read_to_string(cgroups.join("cgroup.controllers"))
            .unwrap()
            .split_whitespace()
            .any(|name| name == "pids")
    );
    let dir = Dir::new();
    struct Group(PathBuf);
    impl Drop for Group {
        fn drop(&mut self) {
            let _ = fs::write(self.0.join("cgroup.kill"), "1\n");
            let _ = fs::remove_dir(&self.0);
        }
    }
    let group = Group(cgroups.join(dir.path().file_name().unwrap()));
    fs::create_dir(&group.0).unwrap();
    let main = dir.path().join("main");
    let pid = dir.path().join("pid");
    fs::write(&main, "[host]\nfix_bindv6only=false\n").unwrap();
    for limit in [1, 2] {
        assert!(fs::read_to_string(group.0.join("cgroup.procs")).unwrap().is_empty());
        // Populate the child before enabling its controller; domain-invalid
        // children reject migration on kernels used by the container fixtures.
        fs::write(cgroups.join("cgroup.subtree_control"), "-pids\n").unwrap();
        let daemon = start(&main, false, &pid);
        let mut launch = Command::new("timeout");
        launch
            .args([
                "-k", "2", "5",
                "/bin/sh",
                "-c",
                "set -eu; printf '%s\\n' \"$$\" > \"$1/cgroup.procs\"; IFS= read -r current < /proc/self/cgroup; [ \"$current\" = \"0::/${1##*/}\" ]; printf '%s\\n' +pids > /sys/fs/cgroup/cgroup.subtree_control; printf '%s\\n' \"$2\" > \"$1/pids.max\"; shift 2; exec \"$@\"",

                "sh",
            ])
            .arg(&group.0)
            .arg(limit.to_string())
            .arg(daemon.get_program())
            .args(daemon.get_args());
        for (key, value) in daemon.get_envs() {
            match value {
                Some(value) => {
                    launch.env(key, value);
                }
                None => {
                    launch.env_remove(key);
                }
            }
        }
        let stdout = dir.path().join(format!("stdout-{limit}"));
        let stderr = dir.path().join(format!("stderr-{limit}"));
        let status = launch
            .stdout(fs::File::create(&stdout).unwrap())
            .stderr(fs::File::create(&stderr).unwrap())
            .status()
            .unwrap();
        let output = std::process::Output {
            status,
            stdout: fs::read(stdout).unwrap(),
            stderr: fs::read(stderr).unwrap(),
        };
        assert_eq!(output.status.code(), Some(1), "limit={limit}: {output:?}");
        assert!(
            String::from_utf8_lossy(&output.stderr)
                .contains(&io::Error::from_raw_os_error(libc::EAGAIN).to_string()),
            "limit={limit}: {output:?}"
        );
        assert_eq!(
            fs::read_to_string(group.0.join("pids.current"))
                .unwrap()
                .trim(),
            "0"
        );
        assert_eq!(
            fs::read_to_string(group.0.join("pids.peak"))
                .unwrap()
                .trim(),
            limit.to_string()
        );
        assert!(!pid.exists());
        println!("DAEMON_FORK_TASK_LIMIT_CHECKED={limit}");
    }
}
