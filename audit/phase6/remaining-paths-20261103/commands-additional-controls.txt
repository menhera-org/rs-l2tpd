set -eu
root=$PWD
record=$root/audit/phase6/remaining-paths-20261103
candidate=$(cat /tmp/phase6-resolver-validation-path)
cp -Rp "$root/src/." "$candidate/src/"
cp -Rp "$root/tests/." "$candidate/tests/"
cd "$candidate"
export CARGO_BUILD_JOBS=2
git --no-pager diff -- src tests Cargo.toml Cargo.lock build.rs > "$record/additional-control-source-before.patch"

 cp "$root/src/reconciler/events.rs" src/reconciler/events.rs

[ -x "$unit" ]
result=0
docker run --rm --privileged --network none -v "$candidate:$candidate:ro" --entrypoint "$candidate/$unit" nd-test:u2404 --exact reconciler::events::resource_tests::removing_a_member_distinguishes_direct_and_parent_bindings --test-threads=1 --nocapture > "$record/control-direct-member-removal.log" 2>&1 || result=$?
restore
[ "$result" != 0 ] && grep -q 'panicked' "$record/control-direct-member-removal.log"
printf 'REMAINING_NEGATIVE_CONTROL=direct-member-removal REJECTED\n'
