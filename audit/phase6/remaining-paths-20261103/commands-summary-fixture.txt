set -eu
record=$PWD/audit/phase6/remaining-paths-20261103
fixture=$record/coverage-summary-fixture.rs
scratch=$(mktemp -d /tmp/rs-l2tpd-coverage-summary.XXXXXXXX)
sysroot=$(rustc +nightly-2026-10-28 --print sysroot)
llvm=$sysroot/lib/rustlib/x86_64-unknown-linux-gnu/bin
rustc +nightly-2026-10-28 --edition=2024 -D warnings -C instrument-coverage -Z coverage-options=branch "$fixture" -o "$scratch/fixture"
for mode in complementary both; do
 LLVM_PROFILE_FILE="$scratch/$mode.profraw" "$scratch/fixture" "$mode"
 "$llvm/llvm-profdata" merge -sparse "$scratch/$mode.profraw" -o "$scratch/$mode.profdata"
 "$llvm/llvm-cov" export "$scratch/fixture" --instr-profile "$scratch/$mode.profdata" > "$scratch/$mode.json"
 "$llvm/llvm-cov" report "$scratch/fixture" --instr-profile "$scratch/$mode.profdata" --show-functions "$fixture" > "$record/summary-fixture-$mode.log"
 "$llvm/llvm-cov" show "$scratch/fixture" --instr-profile "$scratch/$mode.profdata" --show-branches=count --show-instantiations > "$record/summary-fixture-$mode.txt"
 /home/dev/.local/bin/jq '[.data[0].files[]|{filename,summary,branches,segments}]' "$scratch/$mode.json" > "$record/summary-fixture-$mode.json"
done
printf 'fixture artifacts: %s\n' "$scratch"
for mode in complementary both; do
 if [ "$mode" = complementary ]; then lines=80.00%; branches=50.00%; else lines=100.00%; branches=100.00%; fi
 awk -v lines="$lines" -v branches="$branches" '
  /6choose/ { found++; if ($7 != lines || $10 != branches) exit 1 }
  END { if (found != 2) exit 1 }
 ' "$record/summary-fixture-$mode.log"
done
printf 'SUMMARY_AGGREGATION_FIXTURE_AND_BOTH_ARMS_CONTROL_RESULT=0\n'
