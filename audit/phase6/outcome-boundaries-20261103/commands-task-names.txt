set -eu
[ "$#" -eq 2 ] || exit 2
helper=$1
record=$2
mkdir -p "$record"
record=$(CDPATH='' cd -P "$record" && pwd)
scratch=$(mktemp -d /tmp/rs-l2tpd-task-name-check.XXXXXXXX)
trap 'rm -rf "$scratch"' 0
mkdir "$scratch/bin"
cat > "$scratch/bin/docker" <<'STUB'
#!/bin/sh
set -eu
case $1 in
 image) printf 'sha256:test-image\n'; exit 0 ;;
 create)
  shift
  name=
  pattern=
  while [ "$#" -gt 0 ]; do
   case $1 in
    --name) name=$2; shift 2 ;;
    --env) pattern=$2; shift 2 ;;
    *) shift ;;
   esac
  done
  printf '%s\n' "$name" > "$TASK_NAME_RECORD/name"
  printf '%s\n' "$pattern" > "$TASK_NAME_RECORD/pattern"
  exit 42
  ;;
 rm) exit 0 ;;
 *) exit 90 ;;
esac
STUB
chmod 755 "$scratch/bin/docker"
for role in first second; do
 mkdir "$record/$role"
 TASK_NAME_RECORD=$record/$role PATH=$scratch/bin:$PATH sh "$helper" unused-image /bin/true "$record/$role/run" "$record/$role/profiles" > "$record/$role.log" 2>&1 &
 case $role in first) first=$! ;; second) second=$! ;; esac
done
status=0
wait "$first" || status=$?
test "$status" -eq 42
status=0
wait "$second" || status=$?
test "$status" -eq 42
first_name=$(cat "$record/first/name")
second_name=$(cat "$record/second/name")
test "$first_name" != "$second_name"
for role in first second; do
 name=$(cat "$record/$role/name")
 pid=${name#rs-l2tpd-thread-limit-}
 pid=${pid%-1}
 case $pid in ''|*[!0-9]*) exit 1 ;; esac
 test "$(cat "$record/$role/pattern")" = "LLVM_PROFILE_FILE=/profiles/thread-$pid-1-%p-%m.profraw"
done
printf 'TASK_NAMES_AND_PROFILE_PATTERNS_USE_DISTINCT_PROCESS_IDS=1\n'
