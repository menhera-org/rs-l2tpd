#!/bin/sh
# Instrumented GNU normal profile in a disposable, networkless guest.
set -eu
[ "$#" -eq 4 ] || { echo "usage: $0 NATIVE_LOG KERNEL_ROOT RECORD PROFILES" >&2; exit 2; }
repo=$(CDPATH='' cd -P "$(dirname "$0")/../.." && pwd)
cd "$repo"
native=$1; kernel_root=$2; record=$3; profiles=$4
for path in "$native" "$kernel_root" "$record" "$profiles"; do
 case $path in /*) ;; *) echo 'use absolute paths' >&2; exit 2 ;; esac
 case $path in *,*) echo 'commas in guest paths are unsupported' >&2; exit 2 ;; esac
done
test -f "$native"
test ! -e "$record" && test ! -e "$profiles"
artifact() {
 path=$(awk -v label="Running $1 (" '{
  at=index($0,label)
  if(at) {line=substr($0,at+length(label));sub(/[)]$/, "",line);print line}
 }' "$native")
 case $path in /*) ;; *) path=$repo/$path ;; esac
 if [ ! -f "$path" ] || [ ! -x "$path" ]; then
  echo "missing coverage executable for $1: $path" >&2
  return 1
 fi
 printf '%s\n' "$path"
}
unit=$(artifact 'unittests src/lib.rs')
binary_unit=$(artifact 'unittests src/bin/rs-l2tpd/main.rs')
linux=$(artifact 'tests/kernel/linux.rs')
review=$(artifact 'tests/kernel/review.rs')
process=$(artifact 'tests/process.rs')
scenarios=$(artifact 'tests/scenarios/main.rs')
case $unit in
 "$repo"/*/x86_64-unknown-linux-gnu/debug/*)
  daemon=${unit%%/x86_64-unknown-linux-gnu/debug/*}/x86_64-unknown-linux-gnu/debug/rs-l2tpd ;;
 *) echo 'coverage requires the explicit x86_64-unknown-linux-gnu target' >&2; exit 2 ;;
esac
test -f "$daemon" && test -x "$daemon"
set -- "$kernel_root"/boot/vmlinuz-*
test "$#" -eq 1 && test -f "$1"
kernel=$1
release=${kernel##*/vmlinuz-}
modules=$kernel_root/lib/modules/$release
test -d "$modules" || modules=$kernel_root/usr/lib/modules/$release
config=$kernel_root/boot/config-$release
test -d "$modules" && test -f "$config"
image=rs-l2tpd-coverage:phase6
image_id=$(docker image inspect --format '{{.Id}}' "$image")
digest=${image_id#sha256:}
case $image_id in sha256:*) ;; *) exit 1 ;; esac
case $digest in ''|*[!0-9a-f]*) exit 1 ;; esac
test "${#digest}" -eq 64
mkdir -p "$record" "$profiles"
chmod 777 "$profiles"
printf '%s\n' "$image_id" > "$record/image.log"
base=$repo/target/coverage-guest-image-$digest
guest=$(mktemp -d /tmp/rs-l2tpd-coverage-guest.XXXXXXXX)
container=
cleanup() {
 if [ -n "$container" ]; then docker rm -f "$container" >/dev/null 2>&1 || :; fi
 rm -rf "$guest"
}
trap cleanup 0
trap 'exit 1' HUP INT TERM
if [ ! -f "$base/complete" ]; then
 mkdir -p "$base"
 container=$(docker create --network none "$image")
 docker export --output "$base/root.tar" "$container"
 docker rm "$container" >/dev/null
 container=
 rm -rf "$base/root"
 mkdir "$base/root"
 tar -xf "$base/root.tar" --exclude=dev --no-same-owner -C "$base/root"
 : > "$base/complete"
fi
cp -Rp "$base/root" "$guest/root"
mkdir -p "$guest/root/dev" "$guest/root/profiles" "$guest/root/lib/modules/$release"
cp -Rp "$modules/." "$guest/root/lib/modules/$release/"
for entry in "kernel-unit:$unit" "binary-unit:$binary_unit" "kernel-linux:$linux" "review-kernel:$review" "process-test:$process" "scenarios:$scenarios"; do
 name=${entry%%:*}; executable=${entry#*:}
 cp "$executable" "$guest/root/$name"
 strip --strip-debug "$guest/root/$name"
 printf '%s git-blob: ' "$name" >> "$record/artifacts.log"
 git --no-pager hash-object --no-filters "$executable" >> "$record/artifacts.log"
done
mkdir -p "$guest/root$(dirname "$daemon")"
cp "$daemon" "$guest/root$daemon"
strip --strip-debug "$guest/root$daemon"
baseline=$(sh tests/docker/baseline.sh "$record")
cp "$baseline" "$guest/root/baseline"
cp "$config" "$record/kernel.config"
printf 'normal\n' > "$guest/root/matrix-mode"
cat > "$guest/prelude" <<'PRELUDE'
# Only this disposable guest changes its receive-buffer cap.
depmod -a
for module in virtio_pci 9pnet_virtio 9p; do modprobe "$module"; done
mount -t 9p -o trans=virtio,version=9p2000.L profiles /profiles
printf '262144\n' > /proc/sys/net/core/rmem_max
# LLVM merges profiles with mmap, which the 9p transport cannot provide.
mkdir /tmp/coverage-profiles
chmod 1777 /tmp/coverage-profiles
export LLVM_PROFILE_FILE=/tmp/coverage-profiles/coverage-%p-%m.profraw


coverage_finish() {
 result=$?
 trap - 0
 if [ "$result" -eq 0 ]; then
  run_suite env RS_L2TPD_TEST_REQUEST_IO_FAULT=buffers-force /kernel-unit --ignored --exact kernel::linux::request::io_fault_tests::request_io_failures_are_fallible_and_buffers_degrade --nocapture || result=$?
 fi
 cp /tmp/coverage-profiles/*.profraw /profiles/ || result=1
 sync
 umount /profiles || result=1

 printf 'COVERAGE_GUEST_RESULT=%s\n' "$result"
 poweroff -f
}
trap coverage_finish 0
PRELUDE
awk -v prelude="$guest/prelude" '
 {print}
 $0 == "mode=$(cat /matrix-mode)" {
  while ((getline line < prelude) > 0) print line
  close(prelude); found++
 }
 END {if(found != 1) exit 1}
' tests/kernel/matrix-init.sh > "$guest/root/init"
chmod 755 "$guest/root/init"
(cd "$guest/root" && find . -print | cpio -o -H newc -R 0:0 > "$guest/initramfs.cpio") > "$record/initramfs.log" 2>&1
append='console=ttyS0 rdinit=/init panic=1 rootfstype=ramfs'
if grep -q '^CONFIG_KEXEC_HANDOVER_ENABLE_DEFAULT=y$' "$config"; then append="$append kho=off"; fi
{
 printf 'scope: GNU instrumented normal matrix profile plus buffer-force case\n'
 printf 'kernel git-blob: '; git --no-pager hash-object --no-filters "$kernel"
 printf 'helper git-blob: '; git --no-pager hash-object --no-filters "$0"
 printf 'command: timeout 1800 qemu-system-x86_64 -accel tcg -machine q35 -cpu max -m 2048 -smp 2 -nodefaults -nographic -serial stdio -no-reboot -kernel IMAGE -initrd INITRAMFS -append %s -fsdev local,id=profiles,path=PROFILES,security_model=none -device virtio-9p-pci,fsdev=profiles,mount_tag=profiles\n' "$append"
} > "$record/command.log"
status=0
timeout 1800 qemu-system-x86_64 -accel tcg -machine q35 -cpu max -m 2048 -smp 2 \
 -nodefaults -nographic -serial stdio -no-reboot -kernel "$kernel" \
 -initrd "$guest/initramfs.cpio" -append "$append" \
 -fsdev "local,id=profiles,path=$profiles,security_model=none" \
 -device virtio-9p-pci,fsdev=profiles,mount_tag=profiles > "$record/run.log" 2>&1 || status=$?
tr -d '\r' < "$record/run.log" > "$record/run.tmp"
mv "$record/run.tmp" "$record/run.log"
grep -aE 'GUEST_KERNEL_RELEASE|test result:|_RESULT=' "$record/run.log" || :
test "$status" -eq 0
grep -qx "GUEST_KERNEL_RELEASE=$release" "$record/run.log"
grep -qx 'CONFORMANCE_RESULT=0' "$record/run.log"
grep -qx 'SCENARIO_RESULT=0' "$record/run.log"
grep -qx 'COVERAGE_GUEST_RESULT=0' "$record/run.log"
grep -q 'REQUEST_IO_FAULT_CHECKED=buffers-force$' "$record/run.log"
if grep -q 'LLVM Profile Error' "$record/run.log"; then exit 1; fi
count=$(find "$profiles" -type f -name '*.profraw' -size +0c -print | awk 'END {print NR}')
test "$count" -gt 0
printf 'NONEMPTY_COVERAGE_GUEST_PROFILES=%s\nCOVERAGE_GUEST_RESULT=0\n' "$count"
