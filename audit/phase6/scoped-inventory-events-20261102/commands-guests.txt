set -eu
root=/home/dev/rs-l2tpd
candidate=$(cat /tmp/phase6-race-validation-path)
record=$root/audit/phase6/scoped-inventory-events-20261102
cd "$candidate"
export CARGO_BUILD_JOBS=2
run() {
 label=$1; image=$2; modules=$3; config=$4
 result=0
 sh tests/kernel/run-guest.sh "$image" "$modules" "$config" binary "$record/kernels/$label" > "$record/guest-$label.log" 2>&1 || result=$?
 printf '%s\n' "$result" > "$record/kernels/$label.exit"
}
mkdir -p "$record/kernels"
noble=$root/target/phase6-kernels/ubuntu-noble-6.8.0-147.147/root
resolute=$root/target/phase6-kernels/ubuntu-resolute-7.0.0-42.42/root
run noble "$noble/boot/vmlinuz-6.8.0-147-generic" "$noble/lib/modules/6.8.0-147-generic" "$noble/boot/config-6.8.0-147-generic" &
run resolute "$resolute/boot/vmlinuz-7.0.0-42-generic" "$resolute/usr/lib/modules/7.0.0-42-generic" "$resolute/boot/config-7.0.0-42-generic" &
for version in 6.8.12 7.0.14 7.4-rc1; do
 tree=$root/target/phase6-kernels/linux-$version-ipv6
 run "vanilla-$version" "$tree/arch/x86/boot/bzImage" - "$tree/.config" &
done
wait
result=0
for exit_file in "$record"/kernels/*.exit; do
 printf '%s=' "${exit_file##*/}"
 cat "$exit_file"
 [ "$(cat "$exit_file")" = 0 ] || result=1
done
exit "$result"
