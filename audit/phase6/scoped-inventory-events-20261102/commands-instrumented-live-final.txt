set -eu
record=$PWD/audit/phase6/scoped-inventory-events-20261102
cargo +nightly-2026-10-28 llvm-cov --locked --all-features --target x86_64-unknown-linux-gnu --lib --branch --no-report > "$record/final-instrumented-library.log" 2>&1
unit=$(awk -v label='Running unittests src/lib.rs (' '{at=index($0,label);if(at){line=substr($0,at+length(label));sub(/[)]$/, "",line);print line}}' "$record/final-instrumented-library.log")
[ -x "$unit" ]
profiles=$(mktemp -d /tmp/rs-l2tpd-scoped-live.XXXXXXXX)
chmod 777 "$profiles"
for repeat in 1 2 3; do
 for test in kernel::linux::l2tp::boundary_tests::filtered_session_inventories_preserve_other_tunnels_and_namespace_ids kernel::linux::creation_deadline_tests::a_timed_out_creation_waits_for_its_capture_and_releases_the_name; do

  docker run --rm --privileged --network none --mount "type=bind,source=$PWD,target=$PWD,readonly" --mount "type=bind,source=$profiles,target=/profiles" --env "LLVM_PROFILE_FILE=/profiles/scoped-$repeat-$name-%p-%m.profraw" --entrypoint "$PWD/$unit" rs-l2tpd-coverage:phase6 --ignored --exact "$test" --nocapture --test-threads=1 >> "$record/instrumented-live-$repeat.log" 2>&1
 done
done
docker run --rm --network none --mount "type=bind,source=$profiles,target=/profiles" --entrypoint sh rs-l2tpd-coverage:phase6 -ec 'chmod a+r /profiles/*.profraw'
for profile in "$profiles/"*.profraw; do
 cp "$profile" "$CARGO_LLVM_COV_TARGET_DIR/${profile##*/}"
done
