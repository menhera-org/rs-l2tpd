set -eu
candidate=$(cat /tmp/phase6-outcome-batch-path)
cd "$candidate"
record=/home/dev/rs-l2tpd/audit/phase6/coverage-line-inventory-20261103
good=$(mktemp /tmp/phase6-checker-line-good.XXXXXXXX)
cp tools/coverage/src/main.rs "$good"
trap 'cp "$good" tools/coverage/src/main.rs' 0
for fault in empty-lines invented-count missing-inventory; do
 cp "$good" tools/coverage/src/main.rs
 case $fault in
 empty-lines)
 old='covered_lines.len() as u64 <= raw_lines
                && (raw_lines == 0 || !covered_lines.is_empty())'
 new='true'
 name=tests::unsupported_or_inconsistent_exports_fail
 ;;
 invented-count)
 old='!covered
                    || executed_region_lines
                        .get(filename)
                        .is_some_and(|lines| lines.contains(line))'
 new='true'
 name=tests::fabricated_covered_lines_without_any_executed_region_fail
 ;;
 missing-inventory)
 old='.keys()
                .copied()
                .collect::<Vec<_>>()'
 new='.keys()
                .copied()
                .filter(|_| false)
                .collect::<Vec<_>>()'
 name=mapping_tests::inventory_rejects_missing_files_functions_and_changed_spans_but_accepts_new_counts
 ;;
 esac
 { printf '%s\n===REPLACE===\n' "$old"; printf '%s' "$new"; } | /tmp/phase6-replace tools/coverage/src/main.rs
 status=0
 cargo test --locked --manifest-path tools/coverage/Cargo.toml "$name" -- --exact --nocapture > "$record/control-$fault.log" 2>&1 || status=$?
 test "$status" -ne 0
 grep -q 'test result: FAILED. 0 passed; 1 failed' "$record/control-$fault.log"
 printf 'CHECKER_CONTROL_REJECTED=%s\n' "$fault"
done
cp "$good" tools/coverage/src/main.rs
printf 'THREE_CHECKER_LINE_CONTROLS_REJECTED=1\n'
