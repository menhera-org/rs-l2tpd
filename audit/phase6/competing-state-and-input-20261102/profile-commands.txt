set -eu
root=$PWD
record=$root/audit/phase6/competing-state-and-input-20261102
collector=$root/target/phase6-coverage-netlink-reply-state-20261101
buildlog=$record/before-timer-strengthening-instrumented-1.log
binary=$(awk '/Running unittests src\/bin\/rs-l2tpd\/main.rs/ {line=$0; sub(/^.*\(/,"",line); sub(/\)$/,"",line); print line; exit}' "$buildlog")
process=$(awk '/Running tests\/process.rs/ {line=$0; sub(/^.*\(/,"",line); sub(/\)$/,"",line); print line; exit}' "$buildlog")
[ -x "$binary" ] && [ -x "$process" ]
case $binary in /*) ;; *) binary=$root/$binary ;; esac
case $process in /*) ;; *) process=$root/$process ;; esac
profiles=$(mktemp -d "$collector/competing-live-XXXXXXXX")
chmod 777 "$profiles"
run_id=${profiles##*/}
for repeat in 1 2 3; do
 docker run --rm --privileged --network none -e "LLVM_PROFILE_FILE=/profiles/$run_id-$repeat-%p-%m.profraw" -v "$root:$root:ro" -v "$profiles:/profiles" --entrypoint /bin/sh nd-test:u2404 -c '
  set -eu
  uname -r
  "$1" --test-threads=1 --nocapture
  "$1" --ignored --exact pidfile::permission_tests::precreated_permissions_links_and_owner --nocapture
  "$1" --ignored --exact pidfile::removal_tests::removal_uses_inode_after_directory_owner_changes --nocapture
  "$1" --ignored --exact files::filesystem_tests::permission_errors_name_unreadable_unlistable_and_uninspectable_paths --nocapture
  "$2" --include-ignored --test-threads=1 --nocapture
 ' sh "$binary" "$process" > "$record/instrumented-container-$repeat.log" 2>&1
done
docker run --rm --network none -v "$profiles:/profiles" --entrypoint /bin/sh nd-test:u2404 -c 'chmod a+r /profiles/*.profraw'
cp "$profiles"/*.profraw "$collector/"
for path in "$profiles"/*.profraw; do
 printf '%s git-blob: ' "${path##*/}"
 git --no-pager hash-object "$path"
done > "$record/profile-transport.log"
printf 'COMPETING_THREE_INSTRUMENTED_LIVE_PROFILES_RESULT=0\n'
