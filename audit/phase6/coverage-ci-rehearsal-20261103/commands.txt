test ! -e "$coverage"
mkdir -p "$record"
cp /tmp/phase6-ci-rehearsal.sh "$record/commands.txt"
git --no-pager rev-parse HEAD > "$record/base-revision"
git --no-pager diff HEAD -- src tests Cargo.toml Cargo.lock > "$record/initial-source.patch"
rustc +nightly-2026-10-28 -vV > "$record/toolchain.log"
export CARGO_LLVM_COV_TARGET_DIR="$coverage"

TEST_SEED=1 cargo +nightly-2026-10-28 llvm-cov --color never --locked --all-features --target x86_64-unknown-linux-gnu --branch --no-report > "$record/native.log" 2>&1
cargo +nightly-2026-10-28 llvm-cov report --target x86_64-unknown-linux-gnu --branch --json --output-path "$coverage/initial-export.json" > "$record/initial-export.log" 2>&1

 TEST_SEED=$seed cargo +nightly-2026-10-28 llvm-cov --color never --locked --all-features --target x86_64-unknown-linux-gnu --branch --no-report > "$record/native-$seed.log" 2>&1
done
sh tools/coverage/run-guest.sh "$record/native.log" "$PWD/target/phase6-kernels/ubuntu-resolute-7.0.0-42.42/root" "$record/guest" "$PWD/target/phase6-ci-rehearsal-guest-profiles" > "$record/guest-driver.log" 2>&1

printf 'CHECKER_EXIT=%s\n' "$status" >> "$record/merged-source-coverage.log"
tail -n 7 "$record/merged-source-coverage.log"
